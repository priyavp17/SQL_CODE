/* 1. Differential backup duration & size */
SELECT
    database_name,
    backup_start_date,
    backup_finish_date,
    DATEDIFF(SECOND, backup_start_date, backup_finish_date) AS Diff_Duration_Seconds,
    CAST(backup_size / 1024.0 / 1024.0 AS DECIMAL(10,2)) AS DiffSize_MB,
    CAST(compressed_backup_size / 1024.0 / 1024.0 AS DECIMAL(10,2)) AS CompressedSize_MB,
    is_compressed
FROM msdb.dbo.backupset
WHERE type = 'I'
ORDER BY backup_start_date DESC;


/* 2. Last successful FULL backup */
SELECT
    database_name,
    MAX(backup_finish_date) AS LastFullBackupTime
FROM msdb.dbo.backupset
WHERE type = 'D'
GROUP BY database_name;


/* 3. Data file size / growth */
SELECT
    DB_NAME(database_id) AS database_name,
    name,
    physical_name,
    size * 8 / 1024 AS DataFileSize_MB
FROM sys.master_files
WHERE type = 0
ORDER BY database_name;


/* 4. Check if backup is waiting on I/O (when running) */
SELECT
    session_id,
    command,
    wait_type,
    wait_time,
    percent_complete
FROM sys.dm_exec_requests
WHERE command LIKE 'BACKUP%';


/* 5. Backup-related wait statistics */
SELECT
    wait_type,
    waiting_tasks_count,
    wait_time_ms
FROM sys.dm_os_wait_stats
WHERE wait_type LIKE 'BACKUP%';


SELECT
    bs.database_name,
    CASE bs.type
        WHEN 'D' THEN 'FULL'
        WHEN 'I' THEN 'DIFF'
        WHEN 'L' THEN 'LOG'
    END AS BackupType,
    bs.backup_start_date,
    bs.backup_finish_date,
    CAST(bs.backup_size / 1024.0 / 1024.0 AS DECIMAL(12,2)) AS BackupSize_MB_BeforeCompression,
    CAST(bs.compressed_backup_size / 1024.0 / 1024.0 AS DECIMAL(12,2)) AS BackupSize_MB_AfterCompression,
    CAST(
        (1 - (bs.compressed_backup_size * 1.0 / NULLIF(bs.backup_size,0))) * 100
        AS DECIMAL(5,2)
    ) AS Compression_Saving_Percentage
FROM msdb.dbo.backupset bs
WHERE bs.backup_start_date >= DATEADD(DAY, -7, GETDATE())
ORDER BY bs.backup_start_date DESC;
SELECT
    bs.database_name,
    CASE bs.type
        WHEN 'D' THEN 'FULL'
        WHEN 'I' THEN 'DIFF'
        WHEN 'L' THEN 'LOG'
    END AS BackupType,
    bs.backup_start_date,
    bs.backup_finish_date,

    -- Before Compression
    CAST(bs.backup_size / 1024.0 / 1024.0 / 1024.0 AS DECIMAL(10,2)) AS BackupSize_GB_BeforeCompression,
    CAST(bs.backup_size / 1024.0 / 1024.0 AS DECIMAL(10,2))           AS BackupSize_MB_BeforeCompression,

    -- After Compression
    CAST(bs.compressed_backup_size / 1024.0 / 1024.0 / 1024.0 AS DECIMAL(10,2)) AS BackupSize_GB_AfterCompression,
    CAST(bs.compressed_backup_size / 1024.0 / 1024.0 AS DECIMAL(10,2))           AS BackupSize_MB_AfterCompression,

    -- Compression %
    CAST(
        (1 - (bs.compressed_backup_size * 1.0 / NULLIF(bs.backup_size, 0))) * 100
        AS DECIMAL(5,2)
    ) AS Compression_Percentage
FROM msdb.dbo.backupset bs
WHERE bs.backup_start_date >= DATEADD(DAY, -7, GETDATE())
ORDER BY bs.backup_start_date DESC;

SELECT
    vs.volume_mount_point AS Drive,
    vs.logical_volume_name AS VolumeName,
    CAST(vs.total_bytes / 1024.0 / 1024 / 1024 AS DECIMAL(10,2)) AS TotalSize_GB,
    CAST(vs.available_bytes / 1024.0 / 1024 / 1024 AS DECIMAL(10,2)) AS FreeSpace_GB,
    CAST((vs.total_bytes - vs.available_bytes) / 1024.0 / 1024 / 1024 AS DECIMAL(10,2)) AS UsedSpace_GB,
    CAST((vs.available_bytes * 100.0 / vs.total_bytes) AS DECIMAL(5,2)) AS FreeSpace_Pct,
    CAST(((vs.total_bytes - vs.available_bytes) * 100.0 / vs.total_bytes) AS DECIMAL(5,2)) AS UsedSpace_Pct
FROM sys.master_files mf
CROSS APPLY sys.dm_os_volume_stats(mf.database_id, mf.file_id) vs
GROUP BY
    vs.volume_mount_point,
    vs.logical_volume_name,
    vs.total_bytes,
    vs.available_bytes
ORDER BY Drive;
